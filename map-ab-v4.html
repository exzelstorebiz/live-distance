// ================= UTIL JARAK ================= function jarakMeter(lat1, lon1, lat2, lon2) { const R = 6371000; // radius bumi (meter) const dLat = (lat2 - lat1) * Math.PI / 180; const dLon = (lon2 - lon1) * Math.PI / 180; const a = Math.sin(dLat / 2) * Math.sin(dLat / 2) + Math.cos(lat1 * Math.PI / 180) * Math.cos(lat2 * Math.PI / 180) * Math.sin(dLon / 2) * Math.sin(dLon / 2); const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a)); return R * c; } function formatJarak(meter) { if (meter < 1000) { return Math.round(meter) + ' m'; } return (meter / 1000).toFixed(1) + ' km'; } // ================= HAPUS TUJUAN ================= function hapusTujuan() { titikTujuan = null; ruteData = null; ruteCoordinates = []; if (markerTujuan) { map.removeLayer(markerTujuan); markerTujuan = null; } if (ruteLayer) { map.removeLayer(ruteLayer); ruteLayer = null; } document.getElementById('tujuanNama').textContent = 'Belum ditentukan'; document.getElementById('topInfo').classList.remove('show'); document.getElementById('instructionBox').classList.remove('show'); document.getElementById('sisaJarak').textContent = '-'; document.getElementById('estimasi').textContent = '-'; document.getElementById('totalJarak').textContent = '-'; document.getElementById('alamatTujuan').value = ''; showToast('Tujuan dihapus'); } // ================= SIMULASI LOKASI UNTUK TESTING ================= // (opsional, hapus jika tidak diperlukan) function simulasiPosisi(lat, lon) { onPosition({ coords: { latitude: lat, longitude: lon, accuracy: 10 } }); } // ================= INIT ================= function init() { // Mulai GPS mulaiGPS(); // Tambah skala peta L.control.scale({ position: 'bottomleft', imperial: false }).addTo(map); // Zoom control di kanan atas (tidak menutupi panel bawah) L.control.zoom({ position: 'topright' }).addTo(map); // Tampilkan pesan selamat datang setTimeout(function() { showToast('Aplikasi siap.
